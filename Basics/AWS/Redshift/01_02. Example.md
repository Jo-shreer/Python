``` hcl
# ============================================================
# PROVIDER.TF
# AWS PROVIDER AND TERRAFORM VERSION
# ============================================================

terraform {
  required_version = ">= 1.6.0"

  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 6.0"
    }
  }
}

provider "aws" {
  region = var.aws_region

  default_tags {
    tags = {
      Environment = "production"
      Project     = "enterprise-data-warehouse"
      ManagedBy   = "terraform"
      Service     = "redshift"
    }
  }
}


# ============================================================
# VARIABLES.TF
# INPUT VARIABLES
# ============================================================

variable "aws_region" {
  description = "AWS deployment region"
  type        = string
  default     = "eu-west-1"
}

variable "cluster_identifier" {
  description = "Redshift cluster identifier"
  type        = string
  default     = "enterprise-prod-redshift"
}

variable "database_name" {
  description = "Initial database name"
  type        = string
  default     = "analytics"
}

variable "master_username" {
  description = "Redshift master username"
  type        = string
  default     = "redshiftadmin"
}

variable "node_type" {
  description = "RA3 node type"
  type        = string
  default     = "ra3.xlplus"
}

variable "number_of_nodes" {
  description = "Number of RA3 nodes"
  type        = number
  default     = 2

  validation {
    condition     = var.number_of_nodes >= 2
    error_message = "Production multi-node clusters require at least 2 nodes."
  }
}

variable "vpc_id" {
  description = "Existing VPC ID"
  type        = string
}

variable "private_subnet_ids" {
  description = "Private subnet IDs across availability zones"
  type        = list(string)
}

variable "application_security_group_ids" {
  description = "Security groups permitted to connect to Redshift"
  type        = list(string)
  default     = []
}

variable "kms_key_arn" {
  description = "Customer managed KMS key ARN"
  type        = string
}

variable "snapshot_retention_days" {
  description = "Automated snapshot retention"
  type        = number
  default     = 14
}

variable "alarm_sns_topic_arn" {
  description = "SNS topic ARN for monitoring alarms"
  type        = string
  default     = null
}


# ============================================================
# LOCALS.TF
# COMMON CONFIGURATION
# ============================================================

locals {
  common_tags = {
    Environment = "production"
    Application = "enterprise-data-warehouse"
    Owner       = "data-platform-team"
  }
}


# ============================================================
# SECURITY.TF
# REDSHIFT SUBNET GROUP
# ============================================================

resource "aws_redshift_subnet_group" "redshift_subnet" {
  name        = "enterprise-prod-redshift-subnet"
  description = "Private subnet group for production Redshift"

  subnet_ids = var.private_subnet_ids

  tags = local.common_tags

  lifecycle {
    precondition {
      condition     = length(var.private_subnet_ids) >= 2
      error_message = "Provide at least two private subnets in different availability zones."
    }
  }
}


# ============================================================
# SECURITY.TF
# REDSHIFT SECURITY GROUP
# ============================================================

resource "aws_security_group" "redshift_sg" {
  name        = "enterprise-prod-redshift-sg"
  description = "Restricted access to production Redshift"
  vpc_id      = var.vpc_id

  tags = local.common_tags
}

# Allow database access only from approved application SGs.

resource "aws_vpc_security_group_ingress_rule" "redshift_ingress" {
  for_each = toset(var.application_security_group_ids)

  security_group_id            = aws_security_group.redshift_sg.id
  referenced_security_group_id = each.value

  ip_protocol = "tcp"
  from_port   = 5439
  to_port     = 5439

  description = "Redshift access from approved application security group"
}

# Restrict outbound access according to enterprise network policy.
# This example permits outbound HTTPS only.
# Add required endpoints and destinations for your architecture.

resource "aws_vpc_security_group_egress_rule" "redshift_https" {
  security_group_id = aws_security_group.redshift_sg.id

  ip_protocol = "tcp"
  from_port   = 443
  to_port     = 443

  cidr_ipv4 = "0.0.0.0/0"

  description = "HTTPS outbound access"
}


# ============================================================
# IAM.TF
# REDSHIFT IAM ROLE
# ============================================================

resource "aws_iam_role" "redshift_role" {
  name = "enterprise-prod-redshift-role"

  assume_role_policy = jsonencode({
    Version = "2012-10-17"

    Statement = [{
      Effect = "Allow"

      Principal = {
        Service = "redshift.amazonaws.com"
      }

      Action = "sts:AssumeRole"
    }]
  })

  tags = local.common_tags
}

# Example least-privilege S3 read policy.
# Replace the bucket and prefix with your actual data locations.

resource "aws_iam_policy" "redshift_s3_policy" {
  name        = "enterprise-prod-redshift-s3-access"
  description = "Restricted S3 read access for Redshift"

  policy = jsonencode({
    Version = "2012-10-17"

    Statement = [
      {
        Effect = "Allow"

        Action = [
          "s3:GetObject"
        ]

        Resource = [
          "arn:aws:s3:::enterprise-data-lake-prod/warehouse/*"
        ]
      },
      {
        Effect = "Allow"

        Action = [
          "s3:ListBucket"
        ]

        Resource = [
          "arn:aws:s3:::enterprise-data-lake-prod"
        ]

        Condition = {
          StringLike = {
            "s3:prefix" = [
              "warehouse/*"
            ]
          }
        }
      }
    ]
  })

  tags = local.common_tags
}

resource "aws_iam_role_policy_attachment" "redshift_s3_attachment" {
  role       = aws_iam_role.redshift_role.name
  policy_arn = aws_iam_policy.redshift_s3_policy.arn
}


# ============================================================
# MAIN.TF
# AMAZON REDSHIFT RA3 CLUSTER
# ============================================================

resource "aws_redshift_cluster" "production" {

  # Cluster identification

  cluster_identifier = var.cluster_identifier

  database_name   = var.database_name
  master_username = var.master_username

  # AWS manages the master password through Secrets Manager.
  manage_master_password = true

  # Compute configuration

  node_type       = var.node_type
  number_of_nodes = var.number_of_nodes
  cluster_type    = "multi-node"

  # Encryption

  encrypted  = true
  kms_key_id = var.kms_key_arn

  # Network security

  publicly_accessible = false

  cluster_subnet_group_name = (
    aws_redshift_subnet_group.redshift_subnet.name
  )

  vpc_security_group_ids = [
    aws_security_group.redshift_sg.id
  ]

  enhanced_vpc_routing = true

  # IAM integration

  iam_roles = [
    aws_iam_role.redshift_role.arn
  ]

  # Automated backups

  automated_snapshot_retention_period = var.snapshot_retention_days

  manual_snapshot_retention_period = 90

  preferred_maintenance_window = "sun:03:00-sun:04:00"

  # Logging

  logging {
    enable               = true
    log_destination_type = "cloudwatch"

    log_exports = [
      "connectionlog",
      "userlog",
      "useractivitylog"
    ]
  }

  # Cluster maintenance

  allow_version_upgrade = true

  maintenance_track_name = "current"

  # Production protection

  deletion_protection = true

  skip_final_snapshot = false

  final_snapshot_identifier = (
    "${var.cluster_identifier}-final-snapshot"
  )

  # Parameter configuration

  cluster_parameter_group_name = (
    aws_redshift_parameter_group.production.name
  )

  tags = local.common_tags

  depends_on = [
    aws_iam_role_policy_attachment.redshift_s3_attachment
  ]

  lifecycle {
    prevent_destroy = true
  }
}

  name   = "enterprise-prod-redshift-params"
  family = "redshift-1.0"

  description = "Enterprise production Redshift parameters"

  parameter {
    name  = "require_ssl"
    value = "true"
  }

  parameter {
    name  = "enable_user_activity_logging"
    value = "true"
  }

  tags = local.common_tags
}


# ============================================================
# MONITORING.TF
# CLOUDWATCH CPU UTILIZATION ALARM
# ============================================================

resource "aws_cloudwatch_metric_alarm" "redshift_cpu" {

  alarm_name = "enterprise-prod-redshift-high-cpu"

  comparison_operator = "GreaterThanThreshold"

  evaluation_periods  = 3
  metric_name         = "CPUUtilization"
  namespace           = "AWS/Redshift"
  period              = 300
  statistic           = "Average"
  threshold           = 80

  alarm_description = "Redshift CPU utilization is above 80%"

  dimensions = {
    ClusterIdentifier = (
      aws_redshift_cluster.production.cluster_identifier
    )
  }

  alarm_actions = (
    var.alarm_sns_topic_arn == null
    ? []
    : [var.alarm_sns_topic_arn]
  )

  tags = local.common_tags
}
```
