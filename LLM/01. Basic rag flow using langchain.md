```python
# STEP 1: Import libraries

from langchain_community.document_loaders import PyPDFLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings, ChatOpenAI
from langchain_community.vectorstores import FAISS
from langchain_core.prompts import ChatPromptTemplate
from dotenv import load_dotenv


# STEP 2: Load API key from .env

load_dotenv()


# STEP 3: Load PDF

loader = PyPDFLoader("example.pdf")
documents = loader.load()

print("Total pages:", len(documents))


# STEP 4: Split PDF into chunks

text_splitter = RecursiveCharacterTextSplitter(
    chunk_size=500,
    chunk_overlap=50
)

chunks = text_splitter.split_documents(documents)

print("Total chunks:", len(chunks))


# STEP 5: Convert chunks into embeddings

embeddings = OpenAIEmbeddings(
    model="text-embedding-3-small"
)


# STEP 6: Store embeddings in FAISS vector database

vectorstore = FAISS.from_documents(
    documents=chunks,
    embedding=embeddings
)


# STEP 7: Create retriever

retriever = vectorstore.as_retriever(
    search_kwargs={"k": 3}
)


# STEP 8: Ask a question

question = "What is this PDF about?"


# STEP 9: Retrieve relevant chunks

relevant_documents = retriever.invoke(question)

context = "\n\n".join(
    document.page_content
    for document in relevant_documents
)

print("Retrieved context:")
print(context)


# STEP 10: Create prompt

prompt = ChatPromptTemplate.from_template("""
Answer the question using only the context below.

If the answer is not in the context, say you don't know.

Context:
{context}

Question:
{question}

Answer:
""")


# STEP 11: Initialize LLM

llm = ChatOpenAI(
    model="gpt-4.1-mini",
    temperature=0
)


# STEP 12: Send context and question to LLM

messages = prompt.invoke({
    "context": context,
    "question": question
})

response = llm.invoke(messages)

```


# STEP 13: Display answer

print("\nFinal Answer:")
print(response.content)
