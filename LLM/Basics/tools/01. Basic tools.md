```
import os
from dotenv import load_dotenv

from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.tools import tool
from langchain.agents import create_agent

load_dotenv()

llm = ChatOpenAI(
    model="gpt-4.1",
    temperature=0
)

prompt = ChatPromptTemplate.from_messages([
    ("system", "You are a helpful customer support assistant."),
    ("human", "{question}")
])

chain = prompt | llm

@tool
def get_order_status(order_id: str) -> str:
    """Get the current status of a customer's order."""

    orders = {
        "ORD123": "Shipped. Expected delivery tomorrow.",
        "ORD456": "Processing. Not yet shipped."
    }

    return orders.get(order_id, "Order not found.")

agent = create_agent(
    model=llm,
    tools=[get_order_status],
    system_prompt="""
    You are an enterprise customer support assistant.
    If the customer asks about order status,
    use the get_order_status tool.
    Never make up order information.
    Answer clearly and professionally.
    """
)

chain_response = chain.invoke({
    "question": "What is an order ID?"
})

print("\nCHAIN RESPONSE:")
print(chain_response.content)

agent_response = agent.invoke({
    "messages": [
        {
            "role": "user",
            "content": "Where is my order ORD123?"
        }
    ]
})

print("\nAGENT RESPONSE:")
print(agent_response["messages"][-1].content)

print("\nTOOL CALLS:")

for message in agent_response["messages"]:
    if message.type == "ai" and message.tool_calls:
        for tool_call in message.tool_calls:
            print("Tool:", tool_call["name"])
            print("Arguments:", tool_call["args"])
```
