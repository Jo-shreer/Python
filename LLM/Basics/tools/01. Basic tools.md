# LangChain + OpenAI Tool-Calling Agent

# Basic tool
```
from langchain_openai import ChatOpenAI
from langchain.agents import create_agent
from langchain.tools import tool


@tool
def get_order_status(order_id: str) -> str:
    """Get the status of a customer order."""

    orders = {
        "ORD123": "Shipped",
        "ORD456": "Processing",
        "ORD789": "Delivered"
    }

    return orders.get(order_id, "Order not found")


llm = ChatOpenAI(
    model="gpt-4.1",
    temperature=0
)


agent = create_agent(
    model=llm,
    tools=[get_order_status],
    system_prompt="You are a customer support assistant."
)


response = agent.invoke({
    "messages": [
        {"role": "user", "content": "Where is my order ORD123?"}
    ]
})



print(response["messages"][-1].content)

for message in response["messages"]:
    print("Type:", message.type)  # human, ai, tool
    print("Content:", message.content)  # Message content
    print("ID:", message.id)  # Message ID
    print("Name:", message.name)  # Tool name
    print("Tool Calls:", message.tool_calls)  # AI tool requests
    print("Invalid Tool Calls:", message.invalid_tool_calls)  # Invalid tool requests
    print("Additional Kwargs:", message.additional_kwargs)  # Extra data
    print("Response Metadata:", message.response_metadata)  # Model metadata
    print("Usage Metadata:", message.usage_metadata)  # Token usage
    print("Tool Call ID:", getattr(message, "tool_call_id", None))  # Tool call ID
    print("Status:", getattr(message, "status", None))  # Message status

```

 Most important attributes for tool calling:
 1. type          - Identifies human, AI, or tool message.
 2. content       - Contains the message or tool result.**
 3. tool_calls    - Shows which tool the AI wants to call and its arguments.
 4. name          - Identifies the tool that executed**.**
 5. tool_call_id  - Connects the tool result to the AI's tool request.

import os

from sqlalchemy import create_engine, String, Integer, select
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, sessionmaker

from langchain_openai import ChatOpenAI
from langchain.agents import create_agent
from langchain.tools import tool


engine = create_engine(
    os.getenv("DATABASE_URL"),
    pool_size=10,
    max_overflow=20,
    pool_pre_ping=True
)

SessionLocal = sessionmaker(
    bind=engine,
    expire_on_commit=False
)


class Base(DeclarativeBase):
    pass


class Order(Base):
    __tablename__ = "orders"

    id: Mapped[int] = mapped_column(Integer, primary_key=True)
    order_id: Mapped[str] = mapped_column(String(50), unique=True)
    status: Mapped[str] = mapped_column(String(50))


class OrderService:

    def get_order_status(self, session, order_id):
        statement = select(Order).where(
            Order.order_id == order_id
        )

        order = session.scalars(statement).first()

        if order:
            return f"Order {order.order_id} status: {order.status}"

        return "Order not found."


order_service = OrderService()


@tool
def get_order_status(order_id: str) -> str:
    """Get the current status of a customer order."""

    try:
        with SessionLocal() as session:
            return order_service.get_order_status(
                session,
                order_id
            )

    except Exception:
        return "Unable to retrieve order information."


llm = ChatOpenAI(
    model="gpt-4.1",
    temperature=0

agent = create_agent(
    model=llm,

    You are an enterprise customer support assistant.

response = agent.invoke({

## 2. Complete Python code

```python
import os
import logging

import httpx
from dotenv import load_dotenv

from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langchain.agents import create_agent

load_dotenv()

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# 1. Configuration

OPENAI_MODEL = os.getenv("OPENAI_MODEL", "gpt-4.1")
ORDER_API_URL = os.environ["ORDER_API_URL"].rstrip("/")
ORDER_API_TOKEN = os.environ["ORDER_API_TOKEN"]


# 2. Initialize OpenAI Chat Model

llm = ChatOpenAI(
    model=OPENAI_MODEL,
    temperature=0,
    timeout=30,
    max_retries=2
)


# 3. Define Tool

@tool
def get_order_status(order_id: str) -> str:
    """Retrieve the current status of a customer order using its order ID."""

    if not order_id or len(order_id) > 50:
        return "Invalid order ID."

    try:
        response = httpx.get(
            f"{ORDER_API_URL}/orders/{order_id}",
            headers={
                "Authorization": f"Bearer {ORDER_API_TOKEN}",
                "Accept": "application/json"
            },
            timeout=10
        )

        if response.status_code == 404:
            return "Order not found."

        response.raise_for_status()

        order_data = response.json()

        return (
            f"Order ID: {order_data['order_id']}, "
            f"Status: {order_data['status']}, "
            f"Expected delivery: "
            f"{order_data.get('expected_delivery', 'Not available')}"
        )

    except httpx.TimeoutException:
        logger.warning("Order API request timed out.")
        return "Order information is temporarily unavailable."

    except httpx.HTTPStatusError:
        logger.exception("Order API returned an HTTP error.")
        return "Unable to retrieve order information."

    except (httpx.RequestError, ValueError, KeyError):
        logger.exception("Order API request or response failed.")
        return "Unable to retrieve order information."


# 4. Create Agent

agent = create_agent(
    model=llm,
    tools=[get_order_status],
    system_prompt="""
    You are an enterprise customer support assistant.

    If the customer asks about order status,
    use the get_order_status tool.

    Never make up order information.
    Answer clearly and professionally.
    """
)


# 5. Invoke Agent

def ask_customer_support(question: str) -> str:

    if not question or not question.strip():
        raise ValueError("Question cannot be empty.")

    response = agent.invoke({
        "messages": [
            {
                "role": "user",
                "content": question.strip()
            }
        ]
    })

    return response["messages"][-1].content


# 6. Execute Application

if __name__ == "__main__":

    question = "Where is my order ORD123?"

    try:
        response = agent.invoke({
            "messages": [
                {
                    "role": "user",
                    "content": question
                }
            ]
        })

        print("\nFINAL RESPONSE:")
        print(response["messages"][-1].content)

        print("\nMESSAGE HISTORY:")

        for index, message in enumerate(response["messages"]):

            print(f"\nMessage {index + 1}")
            print("Type:", message.type)
            print("Content:", message.content)

        print("\nTOOL CALLS:")

        for message in response["messages"]:

            if message.type == "ai" and message.tool_calls:

                for tool_call in message.tool_calls:

                    print("Tool:", tool_call["name"])
                    print("Arguments:", tool_call["args"])
                    print("Tool Call ID:", tool_call["id"])

        print("\nTOOL RESULTS:")

        for message in response["messages"]:

            if message.type == "tool":

                print("Tool:", message.name)
                print("Result:", message.content)
                print("Tool Call ID:", message.tool_call_id)

    except Exception:
        logger.exception("Agent execution failed.")
        print("Sorry, something went wrong. Please try again.")
```

## 3. Explanation of the components

### ChatOpenAI

Initializes the OpenAI model.

```python
llm = ChatOpenAI(
    model="gpt-4.1",
    temperature=0
)
```

The model understands the user's request and decides whether a tool is required.

### @tool

Defines a Python function that the LLM can request.

```python
@tool
def get_order_status(order_id: str) -> str:
    """Retrieve the current status of a customer order."""
```

The function performs the actual API call and returns the order information.

### create_agent()

Connects the LLM with the tools and instructions.

```python
agent = create_agent(
    model=llm,
    tools=[get_order_status],
    system_prompt="You are a customer support assistant."
)
```

The agent manages the model/tool execution loop automatically.

### agent.invoke()

Sends the user's question to the agent.

```python
response = agent.invoke({
    "messages": [
        {
            "role": "user",
            "content": "Where is my order ORD123?"
        }
    ]
})
```

The input contains a dictionary with a list of messages.

### response["messages"]

Contains the conversation and execution history.

Typically:

1. HumanMessage – User's question.
2. AIMessage – Model's tool-call request.
3. ToolMessage – Result returned by the tool.
4. AIMessage – Final answer.

## 5. What are we NOT using?

We are not using these in this implementation:

**ChatPromptTemplate**

Not required because the agent receives instructions through `system_prompt`.

**Traditional LangChain chain**

Not required because the agent manages the model and tool execution loop.

**LLMChain**

Not required because we are using the modern `create_agent()` approach.

## 7. Production considerations

This is a production-oriented starting example. Before deploying, implement:

- Customer authentication.
- Order ownership and authorization checks.
- Secure secrets management.
- API validation and error handling.
- Logging and tracing.
- Conversation history management.
- Tests for unauthorized access and API failures.

Never rely on the LLM to enforce customer data permissions.

## 8. Final takeaway

For this agent-based implementation, remember these five things:

1. ChatOpenAI – Connects to the LLM.
2. @tool – Defines the Python functions.
3. create_agent – Creates the agent.
4. system_prompt – Defines the agent's instructions.
5. agent.invoke – Sends the user's messages.

**The LLM decides, LangChain orchestrates, and your Python tool executes the business operation.**
