from sqlalchemy import String, Integer, ForeignKey
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column


class Base(DeclarativeBase):
    pass


class Order(Base):
    __tablename__ = "orders"

    id: Mapped[int] = mapped_column(Integer, primary_key=True)
    order_id: Mapped[str] = mapped_column(String(50), unique=True)
    customer_id: Mapped[str] = mapped_column(String(100), index=True)
    status: Mapped[str] = mapped_column(String(50))


class ChatSession(Base):
    __tablename__ = "chat_sessions"

    id: Mapped[int] = mapped_column(Integer, primary_key=True)
    thread_id: Mapped[str] = mapped_column(String(100), unique=True)
    customer_id: Mapped[str] = mapped_column(String(100), index=True)

from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from jose import jwt, JWTError

security = HTTPBearer()


def get_current_customer(
    credentials: HTTPAuthorizationCredentials = Depends(security)
):
    try:
        payload = jwt.decode(
            credentials.credentials,
            JWT_SECRET,
            algorithms=[JWT_ALGORITHM]
        )

        customer_id = payload.get("sub")

        if not customer_id:
            raise HTTPException(
                status_code=401,
                detail="Invalid token"
            )

        return customer_id

    except JWTError:
        raise HTTPException(
            status_code=401,
            detail="Invalid authentication token"
        )

from fastapi import HTTPException


def verify_session_owner(db, thread_id, customer_id):

    session = db.query(ChatSession).filter_by(
        thread_id=thread_id,
        customer_id=customer_id
    ).first()

    if not session:
        raise HTTPException(
            status_code=403,
            detail="Access denied"
        )

from sqlalchemy import select

class OrderService:

    async def get_order_status(self, session, order_id,customer_id):

        cache_key = f"order:{customer_id}:{order_id}"
        cached_status = await cache.get(cache_key)

        if cached_status is not None:
            return f"Order {order_id} status: {cached_status}"

        statement = select(Order).where(
            Order.order_id == order_id,
            Order.customer_id == customer_id
        )

        result = await session.execute(statement)
        order = result.scalar_one_or_none()

        if not order:
            return "Order not found."

        await cache.setex(cache_key, 60, order.status)

        return f"Order {order_id} status: {order.status}"


order_service = OrderService()

from langchain.tools import tool
from contextvars import ContextVar

current_customer_id = ContextVar("current_customer_id")


@tool
def get_order_status(order_id: str) -> str:
    """Get the authenticated customer's order status."""

    try:
        customer_id = current_customer_id.get()

        with SessionLocal() as session:
            return order_service.get_order_status(
                session,
                order_id,
                customer_id
            )

    except Exception:
        return "Unable to retrieve order information."

def chat(message: str, thread_id: str, customer_id: str):

    config = {
        "configurable": {
            "thread_id": thread_id
        }
    }

    token = current_customer_id.set(customer_id)

    try:
        response = agent.invoke(
            {
                "messages": [
                    {"role": "user", "content": message}
                ]
            },
            config=config
        )

        return response["messages"][-1].content

    finally:
        current_customer_id.reset(token)

from fastapi import APIRouter, Depends, HTTPException, status
from pydantic import BaseModel, Field
from sqlalchemy.orm import Session
from uuid import uuid4

router = APIRouter()


class ChatSessionResponse(BaseModel):
    thread_id: str


class ChatRequest(BaseModel):
    thread_id: str
    message: str = Field(min_length=1)


class ChatResponse(BaseModel):
    response: str


@router.post(
    "/chat/session",
    response_model=ChatSessionResponse,
    status_code=status.HTTP_201_CREATED
)
def create_session(
    customer_id: str = Depends(get_current_customer),
    db: Session = Depends(get_db)
):

    thread_id = str(uuid4())

    db.add(
        ChatSession(
            thread_id=thread_id,
            customer_id=customer_id
        )
    )

    db.commit()

    return ChatSessionResponse(thread_id=thread_id)


@router.post(
    "/chat",
    response_model=ChatResponse,
    status_code=status.HTTP_200_OK
)
def chat_endpoint(
    request: ChatRequest,
    customer_id: str = Depends(get_current_customer),
    db: Session = Depends(get_db)
):

    verify_session_owner(
        db,
        request.thread_id,
        customer_id
    )

    response = chat(
        request.message,
        request.thread_id,
        customer_id
    )

    return ChatResponse(response=response)
