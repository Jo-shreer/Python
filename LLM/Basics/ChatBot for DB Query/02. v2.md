from datetime import datetime, timedelta, timezone

from jose import jwt
from pwdlib import PasswordHash

from config import (
    JWT_SECRET,
    JWT_ALGORITHM,
    JWT_EXPIRY_MINUTES
)

password_hash = PasswordHash.recommended()


def hash_password(password: str) -> str:

    return password_hash.hash(password)


def verify_password(
    password: str,
    hashed_password: str
) -> bool:

    return password_hash.verify(
        password,
        hashed_password
    )


def create_access_token(customer_id: str) -> str:

    expires_at = datetime.now(timezone.utc) + timedelta(
        minutes=JWT_EXPIRY_MINUTES
    )

    payload = {
        "sub": customer_id,
        "exp": expires_at
    }

    return jwt.encode(
        payload,
        JWT_SECRET,
        algorithm=JWT_ALGORITHM
    )

import logging

from fastapi import Depends, HTTPException
from fastapi.security import (
    HTTPBearer,
    HTTPAuthorizationCredentials
)
from jose import jwt, JWTError
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from config import JWT_SECRET, JWT_ALGORITHM
from database.dependencies import get_db
from database.models import Customer

logger = logging.getLogger("esa.auth")

security = HTTPBearer(auto_error=False)


async def get_current_customer(
    credentials: HTTPAuthorizationCredentials | None = Depends(security),
    db: AsyncSession = Depends(get_db)
):

    if credentials is None:
        raise HTTPException(
            status_code=401,
            detail="Authentication required",
            headers={"WWW-Authenticate": "Bearer"}
        )

    try:
        payload = jwt.decode(
            credentials.credentials,
            JWT_SECRET,
            algorithms=[JWT_ALGORITHM]
        )

        customer_id = payload.get("sub")

        if not customer_id:
            raise HTTPException(
                status_code=401,
                detail="Invalid token"
            )

        result = await db.execute(
            select(Customer).where(
                Customer.id == customer_id
            )
        )

        customer = result.scalar_one_or_none()

        if not customer:
            raise HTTPException(
                status_code=401,
                detail="Customer not found"
            )

        return customer.id

    except JWTError:
        logger.warning("JWT validation failed")

        raise HTTPException(
            status_code=401,
            detail="Invalid or expired token",
            headers={"WWW-Authenticate": "Bearer"}
        )

import logging

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from cache.valkey import cache
from database.models import Order

logger = logging.getLogger("esa.orders")


class OrderService:

    async def get_order_status(
        self,
        session: AsyncSession,
        order_id: str,
        customer_id: str
    ) -> str:

        cache_key = f"order:{customer_id}:{order_id}"

        try:
            cached_status = await cache.get(cache_key)

            if cached_status is not None:
                logger.info("Order cache hit")
                return f"Order {order_id} status: {cached_status}"

        except Exception:
            logger.exception("Valkey read failed")

        statement = select(Order).where(
            Order.order_id == order_id,
            Order.customer_id == customer_id
        )

        result = await session.execute(statement)
        order = result.scalar_one_or_none()

        if not order:
            logger.warning("Order not found for customer")
            return "Order not found."

        try:
            await cache.setex(
                cache_key,
                60,
                order.status
            )

        except Exception:
            logger.exception("Valkey write failed")

        return f"Order {order_id} status: {order.status}"


order_service = OrderService()

import asyncio
import logging

from uuid import uuid4

from fastapi import (
    APIRouter,
    Depends,
    HTTPException,
    Request
)

from pydantic import BaseModel, EmailStr, Field
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from database.dependencies import get_db
from database.models import Customer

from auth.security import (
    hash_password,
    verify_password,
    create_access_token
)

from rate_limit import limiter

logger = logging.getLogger("esa.auth")

router = APIRouter(
    prefix="/auth",
    tags=["Authentication"]
)


class RegisterRequest(BaseModel):
    email: EmailStr
    password: str = Field(min_length=12, max_length=128)


class LoginRequest(BaseModel):
    email: EmailStr
    password: str


class TokenResponse(BaseModel):
    access_token: str
    token_type: str = "bearer"


@router.post("/register", response_model=TokenResponse)
@limiter.limit("3/hour")
async def register(
    request: Request,
    data: RegisterRequest,
    db: AsyncSession = Depends(get_db)
):

    email = str(data.email).strip().lower()

    result = await db.execute(
        select(Customer).where(Customer.email == email)
    )

    if result.scalar_one_or_none():
        raise HTTPException(
            status_code=409,
            detail="Unable to register with this email"
        )

    hashed_password = await asyncio.to_thread(
        hash_password,
        data.password
    )

    customer = Customer(
        id=str(uuid4()),
        email=email,
        password_hash=hashed_password
    )

    db.add(customer)

    try:
        await db.commit()

    except Exception:
        await db.rollback()
        raise

    logger.info("Customer registered")

    return TokenResponse(
        access_token=create_access_token(customer.id)
    )


@router.post("/login", response_model=TokenResponse)
@limiter.limit("5/minute")
async def login(
    request: Request,
    data: LoginRequest,
    db: AsyncSession = Depends(get_db)
):

    email = str(data.email).strip().lower()

    result = await db.execute(
        select(Customer).where(Customer.email == email)
    )

    customer = result.scalar_one_or_none()

    if not customer:
        raise HTTPException(
            status_code=401,
            detail="Invalid email or password"
        )

    password_valid = await asyncio.to_thread(
        verify_password,
        data.password,
        customer.password_hash
    )

    if not password_valid:
        logger.warning("Customer login failed")

        raise HTTPException(
            status_code=401,
            detail="Invalid email or password"
        )

    logger.info("Customer login successful")

    return TokenResponse(
        access_token=create_access_token(customer.id)
    )

import logging

from uuid import uuid4

from fastapi import (
    APIRouter,
    Depends,
    HTTPException,
    Request,
    status
)

from pydantic import BaseModel, Field
from sqlalchemy.ext.asyncio import AsyncSession

from database.dependencies import get_db
from database.models import ChatSession

from auth.dependencies import get_current_customer
from auth.authorization import verify_session_owner

from services.chat_service import chat
from rate_limit import limiter

logger = logging.getLogger("esa.api")

router = APIRouter()


class ChatSessionResponse(BaseModel):
    thread_id: str


class ChatRequest(BaseModel):
    thread_id: str
    message: str = Field(min_length=1, max_length=10000)


class ChatResponse(BaseModel):
    response: str


@router.post(
    "/chat/session",
    response_model=ChatSessionResponse,
    status_code=status.HTTP_201_CREATED
)
@limiter.limit("10/minute")
async def create_session(
    request: Request,
    customer_id: str = Depends(get_current_customer),
    db: AsyncSession = Depends(get_db)
):

    thread_id = str(uuid4())

    db.add(
        ChatSession(
            thread_id=thread_id,
            customer_id=customer_id
        )
    )

    await db.commit()

    logger.info("Chat session created")

    return ChatSessionResponse(
        thread_id=thread_id
    )


@router.post(
    "/chat",
    response_model=ChatResponse
)
@limiter.limit("30/minute")
async def chat_endpoint(
    http_request: Request,
    request: ChatRequest,
    customer_id: str = Depends(get_current_customer),
    db: AsyncSession = Depends(get_db)
):

    await verify_session_owner(
        db,
        request.thread_id,
        customer_id
    )

    try:
        response = await chat(
            request.message,
            request.thread_id,
            customer_id
        )

        logger.info("Chat request completed")

        return ChatResponse(response=response)

    except Exception:
        logger.exception("Chat processing failed")

        raise HTTPException(
            status_code=500,
            detail="Unable to process chat request"
        )

import logging
import time

from uuid import uuid4
from contextlib import asynccontextmanager

from fastapi import FastAPI, Request

from slowapi.errors import RateLimitExceeded
from slowapi import _rate_limit_exceeded_handler

from api.auth import router as auth_router
from api.chat import router as chat_router

from rate_limit import limiter
from cache.valkey import cache
from database.connection import engine

logger = logging.getLogger("esa")


@asynccontextmanager
async def lifespan(app: FastAPI):

    yield

    await cache.aclose()
    await engine.dispose()


app = FastAPI(
    title="Enterprise Support Agent",
    version="2.1.0",
    lifespan=lifespan
)

app.state.limiter = limiter

app.add_exception_handler(
    RateLimitExceeded,
    _rate_limit_exceeded_handler
)


@app.middleware("http")
async def log_requests(request: Request, call_next):

    request_id = str(uuid4())
    start_time = time.perf_counter()

    try:
        response = await call_next(request)

        duration = time.perf_counter() - start_time

        logger.info(
            "HTTP request | id=%s | method=%s | path=%s "
            "| status=%s | duration=%.3fs",
            request_id,
            request.method,
            request.url.path,
            response.status_code,
            duration
        )

        response.headers["X-Request-ID"] = request_id

        return response

    except Exception:
        logger.exception(
            "HTTP request failed | id=%s",
            request_id
        )
        raise


app.include_router(auth_router)
app.include_router(chat_router)

import asyncio
import os

from logging.config import fileConfig

from alembic import context

from sqlalchemy import pool
from sqlalchemy.ext.asyncio import async_engine_from_config

from database.models import Base
from database import models

config = context.config

if config.config_file_name:
    fileConfig(config.config_file_name)

config.set_main_option(
    "sqlalchemy.url",
    os.environ["DATABASE_URL"]
    .replace("postgresql://", "postgresql+asyncpg://", 1)
    .replace("%", "%%")
)

target_metadata = Base.metadata


def run_migrations_offline():

    context.configure(
        url=config.get_main_option("sqlalchemy.url"),
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"}
    )

    with context.begin_transaction():
        context.run_migrations()


def do_run_migrations(connection):

    context.configure(
        connection=connection,
        target_metadata=target_metadata
    )

    with context.begin_transaction():
        context.run_migrations()


async def run_migrations_online():

    connectable = async_engine_from_config(
        config.get_section(config.config_ini_section),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool
    )

    async with connectable.connect() as connection:

        await connection.run_sync(do_run_migrations)

    await connectable.dispose()


if context.is_offline_mode():
    run_migrations_offline()
else:
    asyncio.run(run_migrations_online())
