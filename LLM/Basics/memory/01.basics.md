```
import os
import valkey

from uuid import uuid4
from fastapi import FastAPI
from pydantic import BaseModel

from sqlalchemy import create_engine, String, Integer, select
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, sessionmaker

from langchain_openai import ChatOpenAI
from langchain.agents import create_agent
from langchain.tools import tool
from langgraph.checkpoint.memory import InMemorySaver


engine = create_engine(
    os.getenv("DATABASE_URL"),
    pool_size=10,
    max_overflow=20,
    pool_pre_ping=True
)

SessionLocal = sessionmaker(
    bind=engine,
    expire_on_commit=False
)

cache = valkey.Valkey(
    host=os.getenv("VALKEY_HOST"),
    port=int(os.getenv("VALKEY_PORT", 6379)),
    password=os.getenv("VALKEY_PASSWORD"),
    ssl=True,
    decode_responses=True
)


class Base(DeclarativeBase):
    pass


class Order(Base):
    __tablename__ = "orders"

    id: Mapped[int] = mapped_column(Integer, primary_key=True)
    order_id: Mapped[str] = mapped_column(String(50), unique=True)
    status: Mapped[str] = mapped_column(String(50))


class OrderService:

    def get_order_status(self, session, order_id):
        cached_status = cache.get(order_id)

        if cached_status is not None:
            return f"Order {order_id} status: {cached_status}"

        statement = select(Order).where(
            Order.order_id == order_id
        )

        order = session.scalars(statement).first()

        if order:
            cache.setex(order_id, 60, order.status)
            return f"Order {order_id} status: {order.status}"

        return "Order not found."


order_service = OrderService()


@tool
def get_order_status(order_id: str) -> str:
    """Get the current status of a customer order."""

    try:
        with SessionLocal() as session:
            return order_service.get_order_status(session, order_id)
    except Exception:
        return "Unable to retrieve order information."


llm = ChatOpenAI(
    model="gpt-4.1",
    temperature=0
)

memory = InMemorySaver()

agent = create_agent(
    model=llm,
    tools=[get_order_status],
    checkpointer=memory,
    system_prompt="""
    You are an enterprise customer support assistant.
    Use the get_order_status tool when customers ask about orders.
    Use conversation history to understand follow-up questions.
    Never make up order information.
    Answer clearly and professionally.
    """
)


def chat(message: str, thread_id: str):

    config = {
        "configurable": {
            "thread_id": thread_id
        }
    }

    response = agent.invoke(
        {
            "messages": [
                {"role": "user", "content": message}
            ]
        },
        config=config
    )

    return response["messages"][-1].content


app = FastAPI()


class ChatRequest(BaseModel):
    thread_id: str
    message: str


@app.post("/chat/session")
def create_chat_session():
    return {"thread_id": str(uuid4())}


@app.post("/chat")
def chat_endpoint(request: ChatRequest):
    response = chat(request.message, request.thread_id)
    return {"response": response}
```
